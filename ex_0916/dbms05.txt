1) like 연산자(유사검색)
-사용자가 지정한 문자형태와 일치 또는 포함하고 있는 자원을 검색

% : 모든 값 
_ : 하나의 값

예1) 'a%' : a로 시작하는 모든 자원
예2) '%a' : a로 끝나는 모든 자원
예3) '%a%' : 값 어디든 a가 포함된 모든 자원
예4) 'a____'(언더바4개) : a로 시작하는 값 중에 길이가 5인 자원

 -- 이름이 D로 시작하는 사원들의 사번, 이름을 출력
select employee_id,
	first_name
from employees
where first_name Like 'D%';

-- 이름이 d로 끝나는 사원들의 사번, 이름, 성을 출력
select employee_id,
	first_name,
	last_name
from employees
where first_name Like '%d';

-- 이름의 세번째 글자가 a인 사원들의 사번, 이름을 출력
select employee_id,
	first_name
from employees
where first_name like '__a%';

-- 이름이 여섯글자인 사원중에 네번째 글자가 a인 직원들의 정보를 사번, 이름 순으로 출력
select employee_id,
	first_name
from employees
where first_name like '___a__';

-- 이름의 시작이 s , 끝이 n인 직원들의 정보를 사번, 이름, 직종순으로 출력
select employee_id,
	first_name,
	job_id
from employees
where first_name like 'S%n';

-- 직종에 SA, IT가 포함되어 있지 않은 사원들의 정보를 사번, 이름, 직종순으로 출력
select employee_id,
	first_name,
	job_id
from employees
where job_id NOT LIKE 'SA%' 
and job_id NOT LIKE 'IT%';

-- 이름이 H로 시작하면서 여섯글자 이상인 사원들의 정보를 이름, 직종 순으로 출력
select first_name,
	job_id
from employees
where first_name like 'H_____%';

-- 이름에 소문자 o가 포함되어 있으면서 a로 끝나는 사원들의 사번, 이름을 출력
select employee_id,
	first_name
from employees
where first_name like '%o%a';

-- 전화번호가 011로 시작하는 사원들의 이름, 전화번호를 출력
select first_name,
	phone_numbe
from employees
where phone_number like '011%';

-- 전화번호가 6으로 시작해서 9로 끝나면서 중간에 2가 들어가는 사원들의 정보를 이름, 전화번호 순으로 조회
select first_name,
	phone_number
from employees
where phone_number like '6%2%9';

-- 전화번호에 44가 포함되어 있는 사원들의 이름, 전화번호를 조회
select first_name,
	phone_number
from employees
where phone_number like '%44%';

-- 입사년도에 상관없이 1월에 입사한 모든 사원들의 정보를 사번, 이름, 입사일 순으로 출력
select employee_id,
	first_name,
	hire_date
from employees
where hire_date like '01%';

-- 전화번호에 44가 포함되어 있지 않은 사원들의 정보를 이름, 전화번호 순으로 조회
select first_name,
	phone_number
from employees
where phone_number not like '%44%';

-- 44가 전화번호 중간에만 들어있는 사원들의 이름, 전화번호를 조회
select first_name,
	phone_number
from employees
where phone_number like '%44%'
and phone_number not like '44%'
and phone_number not like '%44';
