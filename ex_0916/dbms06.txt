1) order by(정렬)
-질의 결과에 반환되는 행들을 정렬하고자 할 때
-order by절은 select절의 마지막에 기술

오름차순 정렬 : ASC(생략가능)
내림차순 정렬 : DESC(생략불가)

-- 급여를 많이 받는 사람순으로 사번, 이름, 급여, 입사일순으로 출력
	단, 급여가 같을 경우 입사일이 빠른 사원부터 조회
select employee_id,
	first_name,
	salary,
	hire_date
from employees
order by salary DESC, 
hire_date ASC; ←ASC 생략 가능

-- 월급이 15000이상인 사원들의 사번, 이름, 월급, 입사일 순으로 조회(입사일 빠른 순으로)
select employee_id,
	first_name,
	salary,
	hire_date
from employees
where salary >= 15000
order by hire_date;

-- 입사일이 빠른순으로 이름, 입사일을 출력하되, 입사일이 같다면 이름 내림차순으로 정렬
select first_name,
	hire_date
from employees
order by hire_date,
first_name DESC;

-- 부서번호가 작은 순으로, 부서번호가 같다면 급여 내림차순으로 이름, 부서번호, 급여를 조회
select first_name,
	department_id,
	salary
from employees
order by department_id,
salary DESC;

-- 부서번호 빠른순, 직종이 빠른순, 월급을 많이 받는 사원 순으로 이름, 부서번호, 직종, 월급을 조회
select first_name,
	department_id,
	job_id,
	salary
from employees
order by department_id,
job_id, salary DESC;

-- 직종이 st_clerk이 아닌 사람중에 사번, 이름, 직종, 급여순으로 조회하되 직종 내림차순 정렬
select employee_id,
	first_name,
	job_id,
	salary
from employees
where not job_id = 'ST_CLERK'
order by job_id DESC;

-- 이름에 소문자 s가 포함되어 있지 않은 사원의 이름을 오름차순으로 정렬하여 조회
select first_name
from employees
where first_name not like '%s%'
order by first_name;

-- 이름에 a가 포함되어 있고 월급이 5000이상인 사원들의 사번, 이름, 급여, 부서번호를 조회
	1. 부서번호 작은 순 2. 같은 부서라면, 월급을 많이 받는 순 3. 월급까지 같다면 이름의 알파벳 순
select employee_id,
	first_name,
	salary,
	department_id
from employees
where first_name like '%a%'
and salary >= 5000
order by department_id,
salary DESC, first_name;

-- 성이 n으로 끝나는 사원들의 사번, 이름, 성, 급여를 출력 
	1. 급여가 높은순 2. 성의 알파벳순
select employee_id,
	first_name,
	last_name,
	salary
from employees
where last_name like '%n'
order by salary DESC, last_name;

-- 이름에 a가 포함되어 있고, 성에 s가 포함되어 있는 사원들을 사번, 이름, 성, 급여순으로 출력
	1. 급여가 높은순 2. 이름 알파벳순 3. 성 알파벳순
select employee_id,
	first_name,
	last_name,
	salary
from employees
where first_name like '%a%'
and last_name like '%s%'
order by salary DESC, first_name, last_name;

-- 이름에 a가 포함된 사원중 2005년 이후에 입사한 사원들의 사번, 이름, 입사일, 월급을 조회
	1. 입사일 빠른순 2.급여가 높은순
select employee_id,
	first_name,
	hire_date,
	salary
from employees
where first_name like '%a%'
and hire_date >= '01/01/2005'
order by hire_date, salary DESC;

-- 이름에 a가 포함되어있고, 성에 s가 포함되어 있지 않으며, 월급이 4000이상인 사원들의 정보를
	사번, 이름, 성, 급여, 부서번호, 직종순으로 출력
	1. 부서번호 낮은순 2.급여 많이 받는순 3.이름 알파벳순
select employee_id,
	first_name,
	last_name,
	salary,
	department_id,
	job_id
from employees
where first_name like '%a%'
and last_name not like '%s%'
and salary >= 4000
order by department_id, salary DESC, first_name;
