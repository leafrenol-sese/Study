1) where절(조건절)
- 특정한 조건에 맞는 내용을 검색하기위해 사용하는 키워드

-- 사원테이블에서 급여가 10000이상인 사원들의 정보를 출력
select * 
from employees
where SALARY >= 10000;

--오라클의 비교연산자
>, >=, <, <=, =, !=(<>:not equal)

-- 월급이 5000 이상인 사원의 사번, 이름, 성, 급여를 출력
select EMPLOYEE_ID,
	FIRST_NAME,
	LAST_NAME,
	SALARY
from employees
where salary >= 5000;

-- 사번이 150을 넘는(초과) 사원들의 사번, 성, 입사일을 조회
select EMPLOYEE_ID,
	LAST_NAME,
	HIRE_DATE
from employees
where employee_id > 150;

-- 성이 smith 사원의 정보를 사번, 성, 급여순으로 출력
select EMPLOYEE_ID,
	LAST_NAME,
	SALARY
from employees
where LAST_NAME = 'Smith';
//문자열 지정 출력일 때는 홑따옴표' ' + 첫문자 대문자 가 필요하다

-- 이름이 마이클인 사원의 사번, 이름, 급여, 부서번호를 조회
select employee_id,
	first_name,
	salary,
	department_id
from employees
where first_name = 'Michael';

-- 직종이 sa_rep인 사원들의 사번, 이름, 이메일, 직종, 급여를 조회
select employee_id,
	first_name,
	email,
	job_id,
	salary
from employees
where JOB_ID = 'SA_REP';
//테이블에 저장되어있는 대소문자 그대로 검색을 해야한다

-- 입사일이 06년 1월 1일 이후(초과)인 사원들의 사번, 이름, 입사일을 조회
select employee_id,
	first_name,
	hire_date
from employees
where hire_date > '01/01/2006';

- 보너스를 받지 않는 사원들의 정보를 이름, 급여, 보너스 순으로 출력
select first_name,
	salary,
	commission_PCT
from employees
where commission_pct is null;
//null값을 확인하고 싶을 때는 '='가 아닌 'is'를 사용, '!='는 'is not'을 사용

- 보너스를 받는 사원들의 정보를 이름, 급여, 보너스 순으로 출력
select first_name,
	salary,
	commission_PCT
from employees
where commission_pct is not null;

-- 상사가 없는 사원들의 사번, 이름, 급여를 출력
select employee_id,
	first_name,
	salary
from employees
where manager_id is null;

2) 다중조건(java와 다름!)
&& → and, || → or : 영문으로 대체
-- 급여가 10000이상 15000미만인 사원들의 사번, 이름, 급여를 출력
select employee_id,
	first_name,
	salary
from employees
where salary >= 10000 and salary < 15000;

-- 급여가 5000미만이거나 20000이상인 사원들의 사번, 이름, 급여, 입사일을 출력
select employee_id,
	first_name,
	salary,
	hire_date
from employees
where salary < 5000 or salary >= 20000;

-- 05년 ~ 06년 사이에 입사한 사원들의 이름, 입사일, 직종을 출력
select first_name,
	hire_date,
	job_id
from employees
where hire_date < '01/01/2007' and hire_date > '12/31/2004';

-- 직종이 sa_man 이거나 it_prog인 사원들의 사번, 이름, 직종을 조회
select employee_id,
	first_name,
	job_id
from employees
where job_id = 'SA_MAN' or job_id = 'IT_PROG';
//여기서는 equal= 을 사용가능, 검색하고자 하는 문자열은 홑따옴표' '를 이용

3-1) in 연산자(or)
- 주어진 목록에서 찾고자 하는 값이 있는 경우

-- in을 사용하지 않았을 때
select employee_id,
	first_name,
	job_id
from employees
where job_id = 'SA_MAN' or 
	job_id = 'IT_PROG' or 
	job_id = 'HP_REP';

-- in을 사용했을 때
select employee_id,
	first_name,
	job_id
from employees
where job_id IN( 'SA_MAN', 'IT_PROG', 'HR_REP' );

-- 사번이 100, 102, 104, 106번인 사원들의 모든 정보 조회
select *
from employees
where employee_id IN ( 100, 102, 104, 106 );

3-2) between 연산자(and)
- A와 B사이의 값을 조회하고자 할 때

-- 월급이 4000~ 8000 사이인 사원들의 사번, 이름, 급여를 출력
select employee_id,
	first_name,
	salary
from employees
where salary >= 4000 and salary <= 8000;
//위아래가 같음!
select employee_id,
	first_name,
	salary
from employees
where salary between 4000 and 8000;

-- 05년도에 입사한 사원들의 사번, 이름, 직종, 입사일을 조회
select employee_id,
	first_name,
	job_id,
	hire_date
from employees
where hire_date between '01/01/2005' and '12/31/2005';
//숫자를 제외한 문자열은 전부 홑따옴표 ' ' 가 필요하다!

-- 급여가 6000 ~ 12000사이 이면서 50번 부서에서 근무하는 사원들의 이름, 급여, 직종을 조회
select first_name,
	salary,
	job_id
from employees
where salary between 6000 and 12000 and department_id = 50;

-- 부서번호가 80이 아닌 부서에서 근무하는 사원들의 사번, 이름, 직종, 부서번호를 조회
select employee_id,
	first_name,
	job_id,
	department_id
from employees
where department_id NOT IN(80);
//(=where not department_id =80;) 
//( = where department_id != 80;) 

-- 07년도 이후에 입사한 사원들의 정보를 사번, 이름, 직종, 입사일 순으로 출력
select employee_id,
	first_name,
	job_id,
	hire_date
from employees
where hire_date >= '01/01/2007';

-- 급여가 2200, 3200, 5000, 6800인 사원들의 정보를 사번, 이름, 직종, 급여순으로 출력
select employee_id,
	first_name,
	job_id,
	salary
from employees
where salary IN (2200, 3200, 5000, 6800);
