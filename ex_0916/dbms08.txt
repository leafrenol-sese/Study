1. 그룹함수
- avg() : 평균을 구하는 함수
- count() : 반환된 행의 갯수를 구하는 함수( null값은 무시한다 )
- max() : 최대값을 구하는 함수
- min() : 최소값을 구하는 함수
- sum() : 합을 구하는 함수
# 그룹함수와 일반 컬럼은 함께 사용할 수 없다.
예) select count(*), 
	     salary
    from employees; 
→ 에러, 그룹화 전이라 사용X

-- 모든 사원의 수를 구하시오
select count(*)
from employees;
//employees의 모든 행의 갯수

select count(employee_id)
from employees;
//employee_id컬럼의 행의 갯수

-- 부서에 속해있는 사원의 수
select count(department_id)
from employees;

-- 보너스를 받고있는 사원의 수
select count(commission_pct)
from employees;

-- 선임이 있는 사원수를 출력
select count(manager_id)
from employees;

-- 누군가의 상사로 일하고 있는 사원수를 출력
select count( DISTINCT manager_id ) →중복제거
from employees;

-- 80번 부서에 속하는 사원들의 월급의 평균을 출력
select avg(salary)
from employees
where department_id = 80;

-- 50번 부서 사원들의 최대 월급, 최소 월급을 조회
select max(salary), min(salary)
from employees
where department_id = 50;

-- 50번 부서 사원들의 월급의 합을 출력
select sum(salary)
from employees
where department_id = 50;

-- it_prog의 수를 출력
select count( * )
from employees
where job_id = 'IT_PROG';

-- sa_rep직종의 사원들의
월급의 평균, 최고월급, 최저월급, 월급의 합을 출력
select avg(salary), max(salary), min(salary), sum(salary)
from employees
where job_id = 'SA_REP';

-- 보너스를 받지 않는 사원수 출력
select count(*)
from employees
where commission_pct is null;

-- 월급 5000이하인 사원들의 평균 월급을 조회
select avg(salary)
from employees
where salary <= 5000;

-- 2005년 이후 입사자의 수를 조회
select count(*)
from employees
where hire_date >= '01/01/2005';

-- 우리회사에 있는 부서 수를 조회
select count(DISTINCT department_id)
from employees;

*nvl() : null값을 원하는 데이터로 교체하는 함수, 일반 함수와 함께 사용 가능
-- 보너스를 받지 않는 모든 사원들의 보너스를 0.1%로 조정
select commission_pct, 
	nvl(commission_pct, 0.1) → commission_pct 컬럼의 rows 중에 null값이 있다면, 0.1을 추가
from employees; 

-- 우리회사에 있는 부서 수를 조회(사장님 포함)
select count(DISTINCT nvl(department_id, 0))
from employees;

*dual(가상테이블) : 함수나 계산식을 테이블의 생성없이 수행하여 볼 수 있도록 하는 가상(임시)의 테이블

2. 숫자함수
- round() : 반올림을 해 주는 함수
select round(.123), round(.567) 
from dual;

select round(.12756783, 2)
from dual;

-- 80번 부서의 월급의 평균을 소수점
select  round( avg(salary), 1)
from employees
where department_id = 80;

-- it_ptog인 사람들의 평균 월급을 소수점 두자리에서 반올림
select  round( avg(salary), 2)
from employees
where job_id = 'IT_PROG';

-trunc() : 잘라내는(버림) 함수

-이름이 A로 시작하는 사원들의 평균 월급을 소수점 1자리에서 절삭해서 조회
select  trunc( avg(salary), 1)
from employees
where first_name like 'A%';

- power() : 제곱을 구하는 함수
select power(4, 2)
from dual;

-mod() : 나머지를 구하는 함수
select mod(7, 4)
from dual;

-- 짝수 사번을 가진 사원들의 사번, 이름, 급여를 출력
select employee_id,
	first_name,
	salary
from employees
where mod(employee_id, 2) = 0;
//where절 집중! 'mod()를 사용, 2로 나눴을때 나머지가 0이 되는 사원번호'

3. 문자함수
- initcap() 첫 문자를 대문자로 변환하는 함수
- 공백이나 / 를 구분자로 인식한다.

select initcap('good morning') → good(공백)morning 이기 때문에 ☞ Good Morning이 된다 ( 슬래시'/'는 그대로 나온다) 
from dual;

- lower() : 대문자 → 소문자
- upper() : 소문자 → 대문자

select lower('Good Morning')
from dual;

select upper('good morning')
from dual;

-- michael로 조회했을 때, 이름이 Michael인 사원의 사번, 이름을 조회
select employee_id,
	first_name
from employees
where lower(first_name) = 'michael'; → 문자함수는 문자열 그리고 컬럼도 가능!

-- 이름의 어디든 m 또는 M이 들어있는 사원의 사번, 이름을 조회
select employee_id,
	first_name
from employees
where lower(first_name) like '%m%'; → 이름을 전부 소문자로 바꾸고 'm'이 있는 이름 검색
//대문자 M은 이미 원본 데이터가 이름 첫글자가 대문자이기 때문에 upper를 사용하지 않아도 된다

- lpad() : 왼쪽의 자릿수를 늘리는 함수
- rpad() : 오른쪽의 자릿수를 늘리는 함수
select lpad('good', 6, '#') → ##good : 4바이트 문자열을 6바이트로 늘리고, 왼쪽에 남은 2바트에는 '#' 을 대입
from dual;
select lpad('good', 8, '안녕') → 안녕good : 4바이트 문자열을 8바이트로 늘리고, 왼쪽에 남은 2바트에는 '안녕' 을 대입, 한글은 2바이트
from dual;
select rpad('good', 6, '@') → good@@ 만일, @대신 안녕을 넣으면 'good안' 만 조회된다. 왼쪽읽기 때문에 2바이트의 '녕'이 출력X
from dual;

- ltrim() : 왼쪽의 문자를 지우는 함수
- rtrim() : 오른쪽의 문자를 지우는 함수
select ltrim('helloh', 'h')
from dual;  →  ellow
select rtrim('helloh', 'h')
from dual;  → hello

- substr() : 특정 문장을 잘라내는 함수
select substr( 'good morning michael', 8, 4 ) : 왼쪽부터 1로 시작, 공백포함 8번째 글자부터 4개의 글자를 잘라낸다
from dual; → rnin
select substr( 'good morning michael', 8 ) : 8번째 글자 이후 모든 글자를 잘라낸다
from dual; → rning michael
select substr( 'good morning michael', -4 ) : 맨 뒤에서부터 4개의 글자를 잘라낸다
from dual; → hael
select substr( 'good morning michael', -7, 4 ) : 맨 뒤에서부터 7번째 글자부터 4개의 글자를 잘라낸다
from dual; → mich

-- 이름, 입사연도만 출력
select first_name "이름",
	substr( hire_date, -4 ) "입사연도"
from employees;

-- 이름, 전화번호, 전화번호 마지막 4자리를 조회
select first_name,
	phone_number,
	substr(phone_number, -4) last4num
from employees;

-- 모든 사원의 사번, 이름, 가린이름 순으로 출력
	(이름의 앞 세글자만 표기, 뒷부분은 *로 변경)
select employee_id,
	first_name,
	rpad( substr(first_name, 1, 3) , 5, '*' ) result
from employees;

- replace() : 원하는 문자를 교체하는 함수
select replace('good morning', 'morning', 'night')
from dual; → good night

- lenght() : 문자열의 길이를 가져오는 함수
select length('hi')
from dual; → 2

-- 이름의 길이가 6을 초과하는 사원들만 사번, 이름을 출력
select employee_id,
	first_name
from employees
where length(first_name) > 6;

-- 전화번호의 길이가 13자 이상인 사원들의 이름, 전화번호를 조회
select first_name,
	phone_number
from employees
where length(phone_number) >= 13;

- concat() : 문자열을 연결하는 함수
- || : 문자열을 연결하는 연산자
ex)
select concat( 'hello', 'hi' ) → hellohi : 연결하고자 하는 문자열 2개까지 이어붙이기 가능
from dual; 
select 'hello' || 'hi' || 'OK' → hellohiOK 두개 이상 연결 가능 
from dual;

-- 사원의 이름과 성을 +로 이어붙여서 조회
select first_name || '+' || last_name fullname
from employees;
select concat(concat(first_name, '+'), last_name) fullname
from employees;

-instr() : 특정 문자열의 위치를 찾아 반환하는 함수
select first_name,
	instr(first_name, 'ar' ) inx  
from employees;  →  Sarah	2(번째부터 ar이 있음)

-- 이름의 마지막 2글자를 **로 대체하여 이름, 가린이름 순으로 출력
ex) Ellen	Ell**
select first_name, 
	rpad( substr( first_name, 1, length(first_name)-2 ) , length(first_name) , '*' ) result
from employees;
ex ) Ellen	E****
→ 오른쪽 자리에 추가 ( 글자 잘라내기(이름, 1~ 이름총길이-2) , (이름 총길이) , 남은 바이트에 '*' 추가) 

-- 이름의 앞 한 글자를 제외한 나머지 글자를 * 대체하여 이름, 가린 이름 순으로 출력
ex ) Ellen	E****
select first_name,
	rpad( substr( first_name, 1, 1) , length(first_name) , '*' ) result
from employees; 
→ substr( first_name, 1, 1) = 1번자리의 한 글자를 추출,  rpad( length(first_name) , '*') 오른쪽에 , 문자 길이만큼, '*'을 채운다
