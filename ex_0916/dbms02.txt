1) 테이블 생성
create table PERSON( 
	--primary key(기본키) : 중복을 허용해서는 안되는 고유한 식별자
	--candidate key(후보키) : id, name과 같이 기본키가 될 자격이 있는 컬럼들
	id NUMBER(2) primary key,
	name VARCHAR2(30) 
);

//사람 테이블을 만들겠다(
	숫자로 된 id(두자리수까지)(id는 중복될 수 없음)
	문자열로 된 이름(30바이트까지) 고유하게 식별하는 컬럼으로
);

//PERSON이라는 테이블을 생성하겠다.
//id라는 컬럼을 만들겠다. → 숫자 데이터를 저장하며, 최대 2자리 정수를 저장할 수 있다. → PRIMARY KEY를 지정하여 기본키로 설정한다.
//name이라는 컬럼을 만들겠다. → 가변 길이 문자열을 저장하며, 최대 30바이트까지 저장할 수 있다.

--테이블에 컬럼 추가
alter table PERSON add birth DATE; 
//날짜 컬럼을 PERSON 테이블에 추가하겠다 → date 형식으로(date 입력은 대소문자 관계없음)

--컬럼명 수정
alter table PERSON rename column birth to birthday;
//PERSON 테이블의 'birth' 컬럼을 'birthday'로 이름을 바꾸겠다

--컬럼속성 수정
alter table PERSON modify name VARCHAR2(50) not null;
//PERSON 테이블의 name 컬럼의 VARCHAR2를 50바이트까지로 수정한다. null은 포함되지 않게.

--컬럼 삭제
alter table PERSON drop column birthday;
//PERSON 테이블의 birthday 컬럼을 삭제하겠다.

--테이블 삭제
drop table PERSON;
//PERSON table을 삭제하겠다.

2) 테이블 생성 문제
--id, pwd, name, email, phone를 저장하는 회원 테이블(member_t)을 만들어보자.
--단, id는 기본키로 지정해야하며, pwd와 name은 null을 허용하지 않도록 한다.
create table MEMBER_T( 
	id NUMBER(2) primary key,
	pwd VARCHAR2(20) not null,
	name VARCHAR2(50) not null,
	email VARCHAR2(50),
	phone VARCHAR2(20)
);

--member_t 테이블에 회원가입날짜를 저장할 수 있는 reg_date컬럼을 추가
alter table MEMBER_T add REG_DATE date;

--이메일 컬럼의 크기를 100으로 수정하기
--전화번호 저장 컬럼을 삭제하기
alter table MEMBER_T modify email VARCHAR2(100);
alter table MEMBER_T drop column PHONE;

--MEMBER_T 테이블 삭제
drop table MEMBER_T;
