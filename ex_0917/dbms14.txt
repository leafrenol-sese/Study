-- 회사에서 (월급을 가장 많이 받는) 사원의 사번, 이름, 월급을 조회
select employee_id,
	first_name,
	salary
from employees
where salary = ( 
	select max(salary)
	from employees
); 

-- (150번 사원과 같은 월급을 받는) 모든 사원들의 정보를 사번, 이름, 급여순으로 출력
select employee_id,
	first_name,
	salary
from employees
where salary = (
	select salary
	from employees 
	where employee_id =150

-- (Bruce와 같은 부서)에서 근무중인 사원들의 사번, 이름을 출력
select employee_id,
	first_name
from employees
where department_id = (
	select department_id
	from employees
	where first_name = 'Bruce'
);

-- (111번의 사원과 같은 직종)이면서 (159번 사원의 급여)를 초과하는 사원들의 정보를

-- 직종별 급여의 평균을 출력
	평균 급여가 (bruce의 급여)를 초과하는 경우만 조회
select job_id,
	avg(salary)
from employees
group by job_id
having avg(salary) > (
	select salary
	from employees
	where first_name = 'Bruce'
);

-- 숙제
(유럽지역 사원들)의 평균 급여보다 많이 받는(초과) 
사원들의 이름, 급여, 지역을 조회
select e.first_name,
	e.salary,
	r.region_name
from employees e
join departments d on e.department_id = d.department_id
	join locations l on d.location_id = l.location_id
	join countries c on l.country_id = c.country_id
	join regions r on c.region_id = r. region_id
where e.salary > (
	 select avg(e2.salary)
	from employees e2
	join departments d2 on e2.department_id = d2.department_id
	join locations l2 on d2.location_id = l2.location_id
	join countries c2 on l2.country_id = c2.country_id
	join regions r2 on c2.region_id = r2.region_id
    where r2.region_name = 'Europe'
);
