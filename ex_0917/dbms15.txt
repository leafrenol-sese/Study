-- employees 테이블을 연습용 테이블에 복사
create table EMPL
AS select * from employees;
→AS는 생략불가!  EMPL이라는 테이블에 employees 테이블의 모든 내용 복사

<필수기입항목> : 성, 이메일, 입사일, 직종

<확인 조회> :
select * from EMPL
where employee_id = 997; 

-- 사번이 999번이고, 이름이 "홍길동" 성이 "Hong"인 사원을 추가 + 이메일, 입사일, 직종
insert into EMPL( employee_id , first_name, last_name, email, hire_date, job_id)
values(999, '홍길동' , 'Hong' , 'hong111' , sysdate, 'prog');

-- 사번 997, 이름'영희', 성'이', 월급 6000, 오늘 입사함
insert into empl(employee_id , first_name, last_name, salary, hire_date, email, job_id)
values(997, '영희', '이', 6000, sysdate, 'lee0', 'IT_PROG');

-- 사번 994, 이름 "장비" 성"익덕" 월급 "7000" 보너스 "0.1" 부서번호 80 입사일은 오늘
